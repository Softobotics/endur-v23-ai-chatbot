 Text content: For example, if the Query Manager is called from the Trading Manager main window, then the Trading Manager browser will show the effect of the script.  The scripts should usually return with an EXIT_Succeed.   EXIT_Fail will cause the query engine to not execute subsequent query pages, but pages already processed (including the page that the script is referenced from) will be saved in the query_result table.  A good use for this script is to pull in the hedge deals for each deal selected in the original query.  A menu-driven text field has been added to the query window, to allow the user to select a script.  When the user saves the query, the selected scripts are stored as part of the query, allowing the same scripts to be loaded back in when the saved query is reloaded. It should be noted here that when a user runs a query with one or more associated scripts, the results are returned in a table called “argt.”  The script passes the following significant columns in the argt table:  Other columns are passed in argt, but not set are as above:  Sample AVS Script /* File Name:           QueryScriptGetHedgesTest.mls Script Type:         Process - do NOT run manually. Parameter Script:    None Display Script:      None  This script will add additional deals (Hedges) whenever a query is executed that calls this script.  */  { // get data from query code query_id = TABLE_GetIntN (argt, "QueryId", 1); page_num = TABLE_GetIntN (argt, "PageNum", 1); do_status = TABLE_GetIntN (argt, "DoStatus", 1);  // get query list from db tran_list = TABLE_New(); where = "unique_id = " + query_id; TABLE_LoadFromDbWithSQL (tran_list, "query_result", "query_result", where); TABLE_SetColName (tran_list, 1, "tran_num");  // Show original query_result list // TABLE_ViewTable (tran_list);  // Go through the list of deals in the original query_result list, and attempt to add associated hedge deals. for (i = 1; i <= TABLE_GetNumRows(tran_list); i++) { hedge_tran_list = TABLE_New(); tran_num = TABLE_GetIntN(tran_list, "tran_num", i); where = "deal_tracking_num = " + tran_num; TABLE_LoadFromDbWithSQL (hedge_tran_list, "hedge_deal_num", "hedge_association", where); for (j = 1; j <= TABLE_GetNumRows(hedge_tran_list); j++) { hedge_deal_num = TABLE_GetIntN (hedge_tran_list, "hedge_deal_num", j); QUERY_Insert (query_id, hedge_deal_num); }  // The following scrap of code will call TABLE_ViewTable for all deals that have hedge deals assoc with them if (1 == 0) { if (TABLE_GetNumRows(hedge_tran_list) > 0) { TABLE_SetColName (hedge_tran_list, 1, STR_IntToStr(tran_num)); TABLE_ViewTable (hedge_tran_list); } }  // Clear table for either next deal, or after last deal TABLE_Destroy(hedge_tran_list); }  EXIT_Succeed(); }.
